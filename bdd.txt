-- Base de datos: El Sobrino 3232 (compatible con MariaDB y MySQL)
-- Importar:  mysql -u root -p < base-de-datos.sql

CREATE DATABASE IF NOT EXISTS sobrino3232
  CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE sobrino3232;

DROP TABLE IF EXISTS arrepentimientos;
DROP TABLE IF EXISTS pedido_items;
DROP TABLE IF EXISTS pedidos;
DROP TABLE IF EXISTS fans;
DROP TABLE IF EXISTS productos;

CREATE TABLE productos (
  id      INT UNSIGNED NOT NULL AUTO_INCREMENT,
  tipo    ENUM('remera','campera','pantalon') NOT NULL,
  nombre  VARCHAR(100) NOT NULL,
  precio  INT UNSIGNED NOT NULL,          -- pesos argentinos
  foto    VARCHAR(255) DEFAULT NULL,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE pedidos (
  id        INT UNSIGNED NOT NULL AUTO_INCREMENT,
  nombre    VARCHAR(100) DEFAULT NULL,
  email     VARCHAR(150) DEFAULT NULL,
  direccion VARCHAR(255) DEFAULT NULL,
  total     INT UNSIGNED NOT NULL,
  estado    ENUM('pagado','arrepentido') NOT NULL DEFAULT 'pagado',
  fecha     TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE pedido_items (
  id          INT UNSIGNED NOT NULL AUTO_INCREMENT,
  pedido_id   INT UNSIGNED NOT NULL,
  producto_id INT UNSIGNED NOT NULL,
  talle       VARCHAR(4) NOT NULL,
  precio      INT UNSIGNED NOT NULL,
  PRIMARY KEY (id),
  CONSTRAINT fk_items_pedido   FOREIGN KEY (pedido_id)   REFERENCES pedidos(id)   ON DELETE CASCADE,
  CONSTRAINT fk_items_producto FOREIGN KEY (producto_id) REFERENCES productos(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE fans (
  id      INT UNSIGNED NOT NULL AUTO_INCREMENT,
  nombre  VARCHAR(100) NOT NULL,
  email   VARCHAR(150) NOT NULL,
  fecha   TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uq_fans_email (email)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE arrepentimientos (
  id        INT UNSIGNED NOT NULL AUTO_INCREMENT,
  pedido_id INT UNSIGNED NOT NULL,
  motivo    TEXT NOT NULL,
  fecha     TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  CONSTRAINT fk_arrep_pedido FOREIGN KEY (pedido_id) REFERENCES pedidos(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
